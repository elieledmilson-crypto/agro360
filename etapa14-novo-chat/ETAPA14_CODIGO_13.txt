// -------------------- Formatadores --------------------

const numberFmt = new Intl.NumberFormat('pt-BR', {
  maximumFractionDigits: 2,
})

function fmtNum(value: number): string {
  return numberFmt.format(value)
}

function fmtKg(value: number): string {
  return `${numberFmt.format(value)} kg`
}

function fmtHa(value: number): string {
  return `${numberFmt.format(value)} ha`
}

function fmtKgHa(value: number): string {
  return `${numberFmt.format(value)} kg/ha`
}

function fmtH(value: number): string {
  return `${numberFmt.format(value)} h`
}

function safeDateLabel(value: string): string {
  try {
    return formatDate(value)
  } catch {
    return value
  }
}

function formatPeriodLabel(period: ReportPeriod): string {
  if (!period.startDate && !period.endDate) {
    return 'Todo o histórico'
  }

  if (period.startDate && period.endDate) {
    return `${safeDateLabel(period.startDate)} a ${safeDateLabel(
      period.endDate,
    )}`
  }

  if (period.startDate) {
    return `Desde ${safeDateLabel(period.startDate)}`
  }

  return `Até ${safeDateLabel(period.endDate as string)}`
}

function BreakdownChart({
  title,
  data,
  formatter,
  emptyMessage,
}: {
  title: string
  data: ReportBreakdownItem[]

}) {
  return (
    <SimpleBarChart

function OverviewTab({
  reportData,
}: {
  reportData: IntegratedReportData
}) {
  const cards: { label: string; value: string }[] = []

  if (reportData.animals) {
    cards.push({
      label: 'Animais ativos',
      value: fmtNum(reportData.animals.active),
    })
  }

  if (reportData.health) {
    cards.push({
      label: 'Vacinações pendentes',
      value: fmtNum(
        reportData.health.pendingVaccinationsCurrent,
      ),
    })
  }

  if (reportData.land) {
    cards.push({
      label: 'Área cadastrada',
      value: fmtHa(reportData.land.totalHectares),
    })
  }

  if (reportData.crops) {
    cards.push({
      label: 'Produção no período',
      value: fmtKg(reportData.crops.totalProductionKg),
    })
  }

  if (reportData.machines) {
    cards.push({
      label: 'Horas trabalhadas no período',
      value: fmtH(reportData.machines.workedHoursInPeriod),
    })
  }

  if (reportData.inventory) {
    cards.push({
      label: 'Itens com estoque baixo',
      value: fmtNum(reportData.inventory.belowMinimumCurrent),
    })
  }

  if (reportData.finance) {
    cards.push({
      label: 'Saldo no período',
      value: formatCurrencyBRL(reportData.finance.balance),
    })
  }

  if (reportData.agenda) {
    cards.push({
      label: 'Atividades atrasadas no período',
      value: fmtNum(reportData.agenda.overdueInPeriod),
    })
  }

  if (cards.length === 0) {
    return (
      <Card className="p-6">
        <h2 className="text-lg font-semibold">
          Nenhum módulo disponível para relatórios
        </h2>
        <p className="mt-2 text-sm text-gray-600 dark:text-gray-400">
          Seu perfil possui acesso ao módulo de Relatórios, mas não
          possui permissão para visualizar dados dos módulos
          operacionais.
        </p>
      </Card>
    )
  }

  return (
    <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
      {cards.map(card => (
        <ReportMetricCard
          key={card.label}
          label={card.label}
          value={card.value}
        />
      ))}
    </div>
  )
}

export default function ReportsPage() {
  const { user } = useAuth()

  const [draftPeriod, setDraftPeriod] = useState<ReportPeriod>({})
  const [appliedPeriod, setAppliedPeriod] = useState<ReportPeriod>(
    {},
  )
  const [activeTab, setActiveTab] = useState<TabKey>('overview')
  const [feedback, setFeedback] = useState<FeedbackType>(null)

  const reportData = useMemo<IntegratedReportData>(
    () => getIntegratedReportData(user, appliedPeriod),
    [user, appliedPeriod],
  )

  const availableTabs = useMemo<TabKey[]>(() => {
    const moduleSet = new Set<string>(reportData.visibleModules)
    return TAB_ORDER.filter(
      tab => tab === 'overview' || moduleSet.has(tab),
    )
  }, [reportData.visibleModules])

  const effectiveTab: TabKey = availableTabs.includes(activeTab)
    ? activeTab
    : 'overview'

  function handleApply() {
    try {
      validateReportPeriod(draftPeriod)
    } catch (error) {
      setFeedback({
        type: 'error',
        message:
          error instanceof Error
            ? error.message
            : 'Período inválido.',
      })
      return
    }

    setFeedback(null)
    setAppliedPeriod({ ...draftPeriod })
  }

  function handleClear() {
    setDraftPeriod({})
    setAppliedPeriod({})
    setFeedback(null)
  }

  return (
    <div className="space-y-6">
      <div className="flex flex-wrap items-center gap-2">
        <h1 className="text-2xl font-bold">Relatórios</h1>

        <HelpTip
          title="Para que servem os Relatórios?"
          description="Os Relatórios consolidam informações já registradas nos módulos do Agro360 para facilitar o acompanhamento da propriedade."
        />
      </div>

      {feedback && (
        <PageFeedback
          type={feedback.type}
          message={feedback.message}
        />
      )}

      <Card className="p-4">
        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-3">
          <label className="flex flex-col text-sm">
            <span className="text-gray-700 dark:text-gray-300 mb-1">
              Data inicial
            </span>
            <input
              type="date"
              value={draftPeriod.startDate ?? ''}
              onChange={event =>
                setDraftPeriod(prev => ({
                  ...prev,
                  startDate: event.target.value || undefined,
                }))
              }
              className="rounded border border-gray-300 dark:border-gray-700 bg-white dark:bg-gray-900 px-3 py-2 text-gray-900 dark:text-gray-100"
            />
          </label>

          <label className="flex flex-col text-sm">
            <span className="text-gray-700 dark:text-gray-300 mb-1">
              Data final
            </span>
            <input
              type="date"
              value={draftPeriod.endDate ?? ''}
              onChange={event =>
                setDraftPeriod(prev => ({
                  ...prev,
                  endDate: event.target.value || undefined,
                }))
              }
              className="rounded border border-gray-300 dark:border-gray-700 bg-white dark:bg-gray-900 px-3 py-2 text-gray-900 dark:text-gray-100"
            />
          </label>

          <div className="flex items-end gap-2 sm:col-span-2">
            <button
              type="button"
              onClick={handleApply}
              className="px-4 py-2 rounded bg-green-600 text-white hover:bg-green-700"
            >
              Aplicar período
            </button>

            <button
              type="button"
              onClick={handleClear}
              className="px-4 py-2 rounded border border-gray-300 dark:border-gray-700 text-gray-700 dark:text-gray-300 hover:bg-gray-100 dark:hover:bg-gray-800"
            >
              Limpar período
            </button>
          </div>
        </div>

        <p className="mt-3 text-sm text-gray-600 dark:text-gray-400">
          Período:{' '}
          <span className="font-medium">
            {formatPeriodLabel(appliedPeriod)}
          </span>
        </p>
      </Card>

      <div className="flex flex-wrap gap-2">
        {availableTabs.map(tab => (
          <button
            key={tab}
            type="button"
            onClick={() => setActiveTab(tab)}
            className={`px-3 py-1.5 rounded-full text-sm border transition ${
              effectiveTab === tab
                ? 'bg-green-600 text-white border-green-600'
                : 'bg-white dark:bg-gray-900 text-gray-700 dark:text-gray-300 border-gray-300 dark:border-gray-700 hover:bg-gray-100 dark:hover:bg-gray-800'
            }`}
          >
            {TAB_LABELS[tab]}
          </button>
        ))}
      </div>

      {effectiveTab === 'overview' && (
        <OverviewTab reportData={reportData} />
      )}

      {effectiveTab === 'animals' && reportData.animals && (
        <AnimalsTab data={reportData.animals} />
      )}

      {effectiveTab === 'health' && reportData.health && (
        <HealthTab data={reportData.health} />
      )}

      {effectiveTab === 'land' && reportData.land && (
        <LandTab data={reportData.land} />
      )}

      {effectiveTab === 'crops' && reportData.crops && (
        <CropsTab data={reportData.crops} />
      )}

      {effectiveTab === 'machines' && reportData.machines && (
        <MachinesTab data={reportData.machines} />
      )}

      {effectiveTab === 'inventory' && reportData.inventory && (
        <InventoryTab data={reportData.inventory} />
      )}

      {effectiveTab === 'finance' && reportData.finance && (
        <FinanceTab data={reportData.finance} />
      )}

      {effectiveTab === 'agenda' && reportData.agenda && (
        <AgendaTab data={reportData.agenda} />
      )}
    </div>
  )
}

import Card from '../ui/Card'

interface BarItem {
  label: string
  value: number
}

interface Props {
  title: string
  data: BarItem[]
  valueFormatter?: (value: number) => string
  emptyMessage?: string
}

export default function SimpleBarChart({
  title,
  data,
  valueFormatter,
  emptyMessage,
}: Props) {
  const format = valueFormatter ?? ((v: number) => String(v))

  const maxValue = data.reduce((max, item) => {
    if (Number.isFinite(item.value) && item.value > max) {
      return item.value
    }

    return max
  }, 0)

  return (
    <Card className="p-4">
      <h3 className="text-sm font-semibold text-gray-900 dark:text-gray-100 mb-3">
        {title}
      </h3>

      {data.length === 0 ? (
        <p className="text-sm text-gray-500 dark:text-gray-400">
          {emptyMessage ?? 'Sem dados.'}
        </p>
      ) : (
        <ul className="space-y-3">
          {data.map(item => {
            const value = Number.isFinite(item.value)
              ? item.value
              : 0

            const pct =
              maxValue > 0
                ? (value / maxValue) * 100
                : 0

            return (
              <li key={item.label}>
                <div className="flex items-center justify-between text-xs mb-1 gap-2">
                  <span className="text-gray-700 dark:text-gray-300 truncate">
                    {item.label}
                  </span>

                  <span className="text-gray-900 dark:text-gray-100 font-medium whitespace-nowrap">
                    {format(value)}
                  </span>
                </div>

                <div className="h-2 rounded bg-gray-100 dark:bg-gray-800 overflow-hidden">
                  <div
                    className="h-full rounded bg-green-500 dark:bg-green-600 transition-all"
                    style={{
                      width: `${pct}%`,
                    }}
                  />
                </div>
              </li>
            )
          })}
        </ul>
      )}
    </Card>
  )
}
